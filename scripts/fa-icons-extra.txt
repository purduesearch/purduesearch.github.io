# Font Awesome icons that a static scan of the source CANNOT see.
#
# The subset build (scripts/build-fa-subset.mjs) only ships glyphs it knows are
# used. Any icon name that is assembled at runtime must be declared here, or it
# will render as a blank box in production.
#
# Format: one per line, either `fa-<name>` (assumed solid) or `<style> fa-<name>`
# where style is fas/far/fab. `#` starts a comment.
#
# Run `npm run build:icons` after editing. The build fails if a name here is not
# a real Font Awesome 6 Free icon, so typos cannot slip through.

# --- Interpolated suffixes: `fas fa-chevron-${open ? 'up' : 'down'}` ---
# TaskModal.jsx, ProjectDetail.jsx, Dashboard.jsx, MeetingNotesView.jsx,
# InventoryPanel.jsx and others build the direction at runtime.
fa-chevron-up
fa-chevron-down
fa-chevron-left
fa-chevron-right

# SafetyBadge.jsx: `fas fa-caret-${open ? 'up' : 'down'}`
fa-caret-up
fa-caret-down

# InsightsTab.jsx: `fas fa-arrow-${isNeutral ? 'right' : isPositive ? 'up' : 'down'}`
fa-arrow-up
fa-arrow-down
fa-arrow-right

# --- Ternary branches inside `fas fa-${...}` ---
# The branch values are bare strings with no `fa-` prefix, so nothing in the

fa-eye
fa-pen
